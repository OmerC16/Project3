import pandas as pd
import os
import numpy as np
import matplotlib.pyplot as plt
import streamlit as st
from matplotlib.patches import Ellipse

SUN_RADIUS = 0.25
W = np.array([6.93764541298151, 14.44410478323698, -154.9549529403448, 332.2918645106256, 436.8382604122162, 198.17179575562477])
bias = -592.2797174702706  # Renamed from 'b' so it doesn't collide with semi-minor axis

st.title("Predict the orbital period by the Asteroid Features")

missDist = st.slider("Enter the miss distance of the asteroid (AU): ", 0.0, 0.5)
minOrbitInter = st.slider("Enter the minimum orbit intersection distance (AU): ", 0.0, 0.48)
eccentricity = st.slider("Enter the orbit's eccentricity: ", 0.0, 0.48)
perihelionDist = st.slider("Enter the Perihelion distance of the asteroid (AU):", 0.18, 1.3)
aphelionDist = st.slider("Enter the Aphelion distance of the asteroid (AU):", 2.01, 8.98)
meanMotion = st.slider("Enter the asteroid's mean motion: ", 0.09, 2.04)

def CalcPrediction(params):
    return params @ W + bias

st.info(
    f"Asteroid's miss distance: {missDist} AU\n\n"
    f"Orbit's minimum intersection distance: {minOrbitInter} AU\n\n"
    f"Orbit's eccentricity: {eccentricity}\n\n"
    f"Perihelion distance: {perihelionDist} AU\n\n"
    f"Aphelion distance: {aphelionDist} AU\n\n"
    f"Mean motion: {meanMotion}"
)

predicted_days = np.round(CalcPrediction(np.array([missDist, minOrbitInter, eccentricity, perihelionDist, aphelionDist, meanMotion])))
st.success(f"The Orbital period is: {predicted_days} days")

# --- ORBITAL GEOMETRY CALCULATIONS ---
# Semi-major axis (a) is half the sum of perihelion and aphelion
a = (aphelionDist + perihelionDist) / 2.0

# Semi-minor axis (b) depends on eccentricity from the slider
b = a * np.sqrt(1 - eccentricity**2)

# Place the Sun (focus) at a fixed coordinate (0, 0)
sun_x, sun_y = 0, 0

# Center of the ellipse is shifted upward from the Sun by (a - perihelionDist)
# so that the bottom tip of the ellipse sits exactly at `perihelionDist` below the Sun
ellipse_center_y = a - perihelionDist

# Create figure and axes
fig, ax = plt.subplots(figsize=(6, 6))

# 1. Draw the Asteroid's Elliptical Orbit
# Because angle=90 rotates the ellipse vertically:
# 'width' corresponds to the vertical major axis (2*a)
# 'height' corresponds to the horizontal minor axis (2*b)
ellipse = Ellipse(
    xy=(sun_x, ellipse_center_y),
    width=2 * a,
    height=2 * b,
    angle=90,
    edgecolor='gray',
    facecolor='none',
    linewidth=2,
    label='Asteroid Orbit'
)
ax.add_patch(ellipse)

# 2. Draw the Sun at the focus (0, 0)
sun = plt.Circle(
    xy=(sun_x, sun_y),
    radius=SUN_RADIUS,
    edgecolor='orange',
    facecolor='gold',
    zorder=5,
    label='Sun (Focus)'
)
ax.add_patch(sun)

# 3. Draw Earth's approximate orbit (1 AU) & Miss Distance indicator for visual context
earth_orbit = plt.Circle((0, 0), radius=1.0, edgecolor='dodgerblue', facecolor='none', linestyle='--', alpha=0.5, label='Earth Orbit (1 AU)')
ax.add_patch(earth_orbit)

if missDist > 0:
    miss_circle = plt.Circle((1.0, 0), radius=missDist, edgecolor='red', facecolor='red', alpha=0.2, label='Miss Distance Zone')
    ax.add_patch(miss_circle)

# Plot perihelion and aphelion markers
ax.plot(0, -perihelionDist, 'go', markersize=5, label=f'Perihelion ({perihelionDist} AU)')
ax.plot(0, aphelionDist, 'ro', markersize=5, label=f'Aphelion ({aphelionDist} AU)')

# Dynamically scale axes to keep the entire orbit centered and visible
max_extent = max(aphelionDist, 1.5) + 1.0
ax.set_xlim(-max_extent, max_extent)
ax.set_ylim(-perihelionDist - 1.5, aphelionDist + 1.0)
ax.set_aspect('equal')
ax.set_title('Asteroid Orbit Simulation (AU)')
ax.legend(loc='upper right', fontsize='small')
ax.grid(True, linestyle=':', alpha=0.4)

# Display in Streamlit
st.pyplot(fig)
